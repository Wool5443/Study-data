\section{Задание}

\begin{enumerate}
    \item Функция $y = f(x)$ задана таблицей:
          \[
              \begin{array}{*{4}{|c}|}
                  \hline
                  X_i & X_1 & X_2 & X_3 \\\hline
                  Y_i & Y_1 & Y_2 & Y_3 \\\hline
              \end{array}
          \]
          Вычислить приближённое значение $y = f(x)$ в точке $x_1 \leq x \leq x_3$ по формуле:
          \[
              f(x) =
              \begin{cases}
                  y_1 + \frac{x - x_1}{x_2 - x_1}(y_2 - y1) \mif
                  x_1 \leq x < x_2 \\
                  y_2 + \frac{x - x_2}{x_3 - x_2}(y_3 - y2) \mif
                  x_2 \leq x < x_3 \\
                  y_3 \mif x = x_3
              \end{cases}
          \]
    \item Дана целочисленная матрица. Вычислить значение по формуле:
          \[
              S = \min_{j=1,n}\max_{i=1,n}(c_{i,j}) \text{, где } c_{i,j}
              \text{ -- элементы матрицы } C[1:n, 1:n].
          \]
    \item Дан целочисленный массив $A[0:n-1]$. К элементам массива
          обращаться при помощи указателя. Написать программу,
          включающую две функции с параметрами. В первой функции
          необходимо подсчитать количество повторений каждого элемента
          массива A. Вторая функция решает следующую задачу:
          сформировать новый массив, содержащий все различные элементы
          массива A по одному разу. Пример:
          \[
              \begin{array}{*{10}{|c}|}
                  \hline
                  A[0:n-1] & 2 & -1 & 0 & 7 & 2 & 7 & 3 & 3 & 2 \\\hline
                  B[0:k-1] & 2 & -1 & 0 & 7 & 3 &   &   &   &   \\\hline
              \end{array}
          \]
\end{enumerate}

\newpage
\section{Постановка задачи}
\given{
    \item $X_1, X_2, X_3, Y_1, Y_2, Y_3$ -- вещ.; $x$ -- вещ.
    \item $C[1:n, 1:n]$ -- цел.
    \item $A[0:n-1]$ -- цел.
}
\result{
    \item $f(x)$ -- вещ
    \item $S$ -- цел.
    \item $counts[0:n-1]$ -- цел.; $B[0:k-1]$ -- цел.
}
\prerequisites{
    \item $x_1 \leq x \leq x_3$
    \item $n \geq 1$
}
\binding{
    \item $f(x) =
        \begin{cases}
            y_1 + \frac{x - x_1}{x_2 - x_1}(y_2 - y_1) \mif
            x_1 \leq x < x_2 \\
            y_2 + \frac{x - x_2}{x_3 - x_2}(y_3 - y_2) \mif
            x_2 \leq x < x_3 \\
            y_3 \mif x = x_3
        \end{cases}$
    \item $S = \min_{j=1,n}\max_{i=1,n}(c_{i,j})$
    \item $\forall i=\overline{0,n-1} \rightarrow counts[i] =
        \cfrac{\sum_{j = 0, A[j] = A[i]}^{n-1}{A[j]}}{A[i]} $;
        $\forall i=\overline{0,n-1} \rightarrow \exists!
        j=\overline{0,k-1}: B[j] = A[i]$
}

\section{Метод решения задачи}

\mathsolution{
    \mathsolutiontask{
        f = y_1 + \frac{x - x_1}{x_2 - x_1}(y_2 - y_1) \mif
            x_1 \leq x < x_2 \\
        f = y_2 + \frac{x - x_2}{x_3 - x_2}(y_3 - y_2) \mif
            x_2 \leq x < x_3 \\
        f = y_3 \mif x = x_3
    }
    \mathsolutiontask{
        max(i) = C[i, 1] \\
        \mathsolutionloop{
            \mfor j = \overline{2, n} \\
            max(i) = C[i, j] \mif C[i, j] > max \\
        } \\
        min = max(1) \\
        \mathsolutionloop{
            \mfor i = \overline{2, n} \\
            t = max(i) \\
            min = t \mif t < min \\
        }
    }
    \mathsolutiontask{
        counts[0:n-1] = \{1\} \\
        \mathsolutionloop{
            \mfor i = \overline{1, n - 1} \\
            \mathsolutionloop{
                \mfor j = \overline{0, i - 1} \\
                (counts[j] = counts[j] + 1, counts[i] = counts[j])
                \mif A[j] = A[i] \\
            } \\
        } \\
        k = 1 \\
        B[0] = A[0] \\
        \mathsolutionloop{
            \mfor i = \overline{1, n - 1} \\
            found = 0 \\
            \mathsolutionloop{
                \mwhile found < k \mand B[found] \neq A[i] \\
                found = found + 1 \\
            } \\
            (B[k] = A[i], k = k + 1) \mif found = k
        }
    }
}

\begin{Verbatim}[commandchars=\\\{\}]

\algline{нач}
    вывод(\algquote{Лаборатоная работа №3})
    вывод(\algquote{Задание №1})

    вывод(\algquote{Введите действительные X1, X2, X3, Y1, Y2, Y3:})

        f := y1 + (x - x1) * (y2 - y1) / (x2 - x1)

    \algelse

    \algifend

\subsection{Программа №3}
\begin{Verbatim}[commandchars=\\\{\}]
\algline{алг} \algquote{Лабораторная работа №3 Задание №3}
\algline{нач}
    вывод(\algquote{Лаборатоная работа №3})
    вывод(\algquote{Задание №3})

    вывод(\algquote{Введите n от 1 до LMAX:})
    \algline{цикл}
        ввод(n)
    \algline{до} 1 \algleq n \algleq LMAX

    вывод(\algquote{Введите целочисленный массив A[0:n-1]:})
    ввод(A[0:n-1])

    \algfor{i := 0}{n - 1}
        counts[i] := 1
    \algforwhileend

    \algfor{i := 1}{n - 1}
        \algfor{j := 0}{i - 1}
            \algif{A[j] = A[i]}
                counts[j] := counts[j] + 1
                counts[i] := counts[j]
            \algifend
        \algforwhileend
    \algforwhileend

    вывод(\algquote{Частоты чисел в массиве:})
    вывод(counts[0:n-1])

    k := 1
    B[0] := A[0]

    \algfor{i := 1}{n - 1}
        found := 0
        \algwhile{found < k \algline{и} B[found] != A[i]}
            found := found + 1
        \algforwhileend
        \algif{found = k}
            B[k] := A[i]
            k := k + 1
        \algifend
    \algforwhileend

    вывод(\algquote{Уникальные числа в массиве в количестве }, k, \algquote{ штук:})
    вывод(B[0:k-1])
\end{Verbatim}
