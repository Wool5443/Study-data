\section{Задание}

\begin{enumerate}
    \item Вычислить массив $r[1:n]$ в соответствии с формулой:
        $r[i] = 1,25 \sin(3ax -ih)$
    \item Из вычисленного массива $r$ удалить все элементны,
        расположенные между последним положительным и первым
        максимальным по модулю элементами.
    \item В полученном массиве $r[1:k]$, где $k$ --- число
        элементов, оставшихся после удаления, подсчитать среднее
        арифметическое элементов, расположенных до последнего
        минимального элемента включительно.
\end{enumerate}

\newpage
\section{Постановка задачи}

\given
{
    \item n --- цел.; a, x, h --- вещ.
    \item Нет входных данных
    \item Нет входных данных
}

\result{
    \item $r[1:n]$ --- вещ.
    \item $r[1:k]$ --- вещ., где $k$ --- длина массива после удаления
        или сообщение <<Нет удалений>>
    \item $\text{average}$ --- вещ.
}

\prerequisites{
    \item $n \in \mathbb{N}, \ n \leq LMAX$, где $LMAX$ ---
        максимальная длинна массива.
}

\binding{
    \item $r[i] = 1,25 \sin(3ax - ih)$
    \item $\exists n_1 : \forall i \in [1, n] \implies |r[i]|
        <= |r[n_1]|, \  \nexists t \in [1, n_1 - 1]: |r[t]| >
        |r[n_1]|$ \\
        $\exists n_2 : r[n_2] > 0 \  \nexists t \in [n_2 + 1, n]:
        r[t] > 0$ \\
        При $n_1 > n_2, \ c = n_1, \ n_1 = n_2, \ n_2 = c$ \\
        $\forall i \in [1, n_1] \cup [n_2, n] \implies \exists t \in
        [1, k]: r'[t] = r[k], \ \forall j \in [1, k] \ \exists q \in
        [1, n_1] \cup [n_2, n]: r'[j] = r[q]$, где $r'$ --- массив
        после удаления.
    \item $\exists km \in [1, k] : \forall i \in [1, k] \implies r[i]
        >= r[km], \ \forall q \in [km + 1, k] \implies r[q] > r[km]$
        \\
        $average = \sum_{i = 1}^{i = km}{r[i]} \cdot \frac{1}{km}$
}

\newpage
\section{Метод решения задачи}
\mathsolution{
    \mathsolutiontask{
        \mathsolutionloop{
            \text{для } i = \overline{1, n} \\
            r[i] = 1,25 \sin(3ax - ih)
        }
    }
    \mathsolutiontask{
        n_1 = 0 \\
        \mathsolutionloop{
            для i = \overline{2, n} \\
            n_1 = i \text{, если } |r[i]| > |r[n_1]| \\
        } \\ \\

        n_2 = n \\
        \mathsolutionloop{
            \text{пока } n_2 > 0 \text{ и } r[n_2] \leq 0 \\
            n_2 = n_2 - 1 \\
        } \\ \\

        (c = n_1, n_1 = n_2, n_2 = c) \text{, если } n_1 > n_2 \\ \\

        m = n_2 - n_1  - 1\\
        k = n - m \text{, если } m > 0 \\
        k = n \text{, в противном случае} \\
        \mathsolutionloop{
            \text{для } i = \overline{n_1 + 1, k} \\
            r[i] = r[i + m] \\
        } \\
    }
    \mathsolutiontask{
        km = 1 \\
        \mathsolutionloop{
            \text{для } i = \overline{2, k} \\
            km = i \text{, если } r[i] \leq r[km]
        } \\ \\

        sum = 0 \\
        \mathsolutionloop{
            \text{для } i = \overline{1, km} \\
            sum = sum + r[i] \\
        } \\
        average = \frac{sum}{km}
    }
}

\section{Описание алгоритма на псевдокоде}

\begin{Verbatim}[commandchars=\\\{\}]
алг \algquote{Лабораторная работа №\labnumber}
    вывод(\algquote{Лабораторная работа №\labnumber})
    вывод(\algquote{Задание №1})

    вывод(\algquote{Введите n от 1 до LMAX:})

    \alguntil
        ввод(n)
    \alguntilend{1 \algleq n \algleq LMAX}

    \alguntil
        ввод(a, x, h)
    \alguntilend{isfinite(a) \algline{и} isfinite(x) \algline{и} isfinite(h)}

    \algfor{i := 1}{n}
        r[i] := 1,25sin(3ax - ih)
    \algforwhileend

    вывод(r[1:n])

    вывод(\algquote{Задание №2})

    n1 := 1
    \algfor{i := 2}{n}
        \algif{|r[i]| > |r[n1]|}
            n1 := i
        \algendif
    \algforwhileend

    n2 := n + 1
    \algfor{i := 1}{n}
        \algif{r[i] > 0}
            n2 := i
        \algendif
    \algline{кц}

    \algif{n1 > n2}
        c := n1
        n1 := n2
        n2 := c
    \algendif

    m := n2 - n1 - 1
    \algif{m > 0}
        k := n - m
    \algelse
        k := n
    \algendif
    \algfor{i := n1 + 1}{k}
        r[i] = r[i + m]
    \algforwhileend

    \algif{k = n}
        вывод(r[1:k])
    \algelse
        вывод(\algquote{Нет удалений})
    \algendif

    km := 1
    \algfor{i := 2}{k}
        \algif{r[i] \algleq r[km]}
            km := i
        \algendif
    \algforwhileend

    sum := 0
    \algfor{i := 1}{km}
        sum := sum + r[i]
    \algforwhileend

    average := sum / km

    вывод(\algquote{Cреднее вышло:}, average)

кон
\end{Verbatim}
